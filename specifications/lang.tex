  \Tp : \Con \to \FJdg \\
  \Tm : (\Gamma : \Con) \to \Tp\,\Gamma \to \FJdg^+ \\
  (-)^{\lock} : \Con \to \Con \\
  (-)^{\key} : \Con \to \Con \\

  \_ : (\Gamma^{\lock})^{\lock} \cong \Gamma^{\lock} \\
  \_ : \Gamma^{\key} \To \Gamma \\
  \_ : \Gamma^{\key} \cong (\Gamma^{\key})^{\key} \\
  \_ : (\Gamma^{\lock})^{\key} \cong \Gamma^{\key} \\
  \Ind : \Con \to \FJdgProp \\
  \_ : (\Gamma : \Con) \to \Ind\,(\Gamma^{\key}) \\
  \init : (\Gamma : \Con) \to \Ind\,\Gamma \to \Tp\,\Gamma^{\lock} \to \Tm\,\Gamma\,A \\
